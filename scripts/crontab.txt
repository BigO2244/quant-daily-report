# Caerus Quant Daily Report — GCP VM Cron Schedule
# Install with: crontab scripts/crontab.txt
#
# All times are America/New_York (handles EST/EDT automatically).
# Runs weekdays only (Mon-Fri = 1-5).
#
# Phase 0b: Claude Research Agent (morning digest) — 6:30 AM ET
# Phase 1:  Precompute (signals, recon, position check) — 7:00 AM ET
# Phase 2:  Order Execution (Alpaca paper API + gated options overlay) — 9:35 AM ET
# Phase 3:  Confirmation (verify trades, send email) — 10:00 AM ET

CRON_TZ=America/New_York
SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin

# Phase 0b: Claude Research Agent (before precompute — 6:30 AM ET)
30 6 * * 1-5 $HOME/quant-daily-report/scripts/cron_research.sh >> $HOME/quant-daily-report/logs/cron_research.log 2>&1

# Security master refresh (artifact-only; before precompute final plan)
45 6 * * 1-5 $HOME/quant-daily-report/scripts/cron_security_master.sh >> $HOME/quant-daily-report/logs/cron_security_master.log 2>&1

# Phase 1: Precompute (consumes research digest)
0 7 * * 1-5 $HOME/quant-daily-report/scripts/cron_precompute.sh >> $HOME/quant-daily-report/logs/cron_precompute.log 2>&1

# Phase 2: Order Execution
35 9 * * 1-5 $HOME/quant-daily-report/scripts/cron_execute.sh >> $HOME/quant-daily-report/logs/cron_execute.log 2>&1

# Phase 2b: LIVE_PILOT automated lane. Disabled unless
# CAERUS_LIVE_PILOT_SCHEDULE_ENABLED=1, and live submission still requires
# CAERUS_LIVE_PILOT_CRON_APPROVED=1 plus CAERUS_LIVE_PILOT_SUBMIT_APPROVED=1.
36 9 * * 1-5 $HOME/quant-daily-report/scripts/cron_live_pilot_execute.sh >> $HOME/quant-daily-report/logs/cron_live_pilot_execute.log 2>&1

# Phase 3b: LIVE_PILOT confirmation email from isolated live-pilot artifacts.
45 9 * * 1-5 $HOME/quant-daily-report/scripts/cron_live_pilot_confirm.sh >> $HOME/quant-daily-report/logs/cron_live_pilot_confirm.log 2>&1

# Phase 3: Confirmation + Email
0 10 * * 1-5 $HOME/quant-daily-report/scripts/cron_confirm.sh >> $HOME/quant-daily-report/logs/cron_confirm.log 2>&1

# Shadow CIO model scorecard email
0 21 * * 1-5 cd $HOME/quant-daily-report && source scripts/runtime_env.sh && activate_runtime_venv $HOME/quant-daily-report && python3 -m scripts.send_shadow_cio_report >> logs/shadow_cio_report.log 2>&1

# Post-close artifact-only price cache hydration for shadow/reporting reads
30 18 * * 1-5 cd $HOME/quant-daily-report && source scripts/runtime_env.sh && activate_runtime_venv $HOME/quant-daily-report && python3 -m scripts.hydrate_price_cache_only --refresh-shadow-artifacts --strict >> logs/price_hydration.cron.log 2>&1

# FR-066: canonical NAV daily build + fail-loud freshness escalation — 7:15 PM ET,
# after price hydration. Read-only telemetry builder; NON_EXECUTIONAL. The
# escalation step always runs (even if the build fails) so silence is never a
# valid state. Installed on the VM 2026-06-10 after owner approval; output is
# captured in logs/portfolio_history.cron.log.
15 19 * * 1-5 cd $HOME/quant-daily-report && source scripts/runtime_env.sh && activate_runtime_venv $HOME/quant-daily-report && ( python3 scripts/build_portfolio_history.py --trade-date $(TZ=America/New_York date +\%F) ; python3 -m core.portfolio_history_escalation --trade-date $(TZ=America/New_York date +\%F) --send ) >> logs/portfolio_history.cron.log 2>&1

# Broker-truth ledger + realized performance + TCA (GET-only broker reads).
45 19 * * 1-5 $HOME/quant-daily-report/scripts/cron_broker_ledger.sh >> $HOME/quant-daily-report/logs/cron_broker_ledger.log 2>&1

# Weekly Model Review (Monday 8:00 AM ET)
0 8 * * 1 $HOME/quant-daily-report/scripts/cron_weekly_review.sh >> $HOME/quant-daily-report/logs/weekly_review.log 2>&1
